% ================================================================
% Licence: CC badge and formal notice
% ================================================================
\cs_new_protected:Npn \__pxtfg_licence_artwork:
  {
    % A recognised CC code selects the bundled PNG; custom licences do not.
    \tl_set:Nx \l_tmpa_tl
      {\tl_lower_case:n {\tl_to_str:V \l__pxtfg_licence_tl}}
    \regex_replace_all:nnN {[^a-z]} {} \l_tmpa_tl
    \str_case:enT {\tl_use:N \l_tmpa_tl}
      {{ccby}{}{ccbync}{}{ccbyncnd}{}{ccbyncsa}{}{ccbynd}{}{ccbysa}{}}
      {
        \clist_map_inline:nn {licenc/,../code/licenc/,tex/code/licenc/}
          {
            \tl_set:Nx \l_tmpb_tl {##1\l_tmpa_tl.png}
            \exp_args:NV \file_if_exist:nT \l_tmpb_tl
              {
                \includegraphics[height=1cm,keepaspectratio]{\l_tmpb_tl}
                \par\medskip
                \clist_map_break:
              }
          }
      }
  }

\cs_new:Npn \__pxtfg_cc_text:n #1
  {
    \str_case:enF {\languagename}
      {
        {catalan}
          {
            Aquesta\space obra\space està\space subjecta\space a\space
            una\space llicència\space de\space
            Reconeixement-NoComercial-CompartirIgual\space #1\space
            \str_if_eq:nnTF {#1}{3.0}{Espanya}{Internacional}
            \space de\space Creative\space Commons
          }
        {spanish}
          {
            Esta\space obra\space está\space sujeta\space a\space
            una\space licencia\space de\space
            Reconocimiento-NoComercial-CompartirIgual\space #1\space
            \str_if_eq:nnTF {#1}{3.0}{España}{Internacional}
            \space de\space Creative\space Commons
          }
        {english}
          {
            This\space work\space is\space licensed\space under\space
            Attribution-NonCommercial-ShareAlike\space #1\space
            \str_if_eq:nnTF {#1}{3.0}{Spain}{International}
            \space by\space Creative\space Commons
          }
      }
      {Creative\space Commons\space CC\space BY-NC-SA\space #1}
  }

\cs_new_protected:Npn \__pxtfg_cc_notice:nn #1#2
  {
    \noindent
    \cs_if_exist:NTF \href
      {
        % Convert expl3 URL punctuation to ordinary characters for hyperref.
        \exp_args:Ne \href {\tl_to_str:n {#2}}
          {\__pxtfg_cc_text:n {#1}}
      }
      {\__pxtfg_cc_text:n {#1}}
    \par
  }

\cs_new_protected:Npn \__pxtfg_licence_content:
  {
    \tl_if_blank:VF \l__pxtfg_licence_tl
      {
        \tl_if_blank:VTF \l__pxtfg_licence_icon_tl
          {\__pxtfg_licence_artwork:}
          {
            \includegraphics[height=1cm,keepaspectratio]
              {\l__pxtfg_licence_icon_tl}
            \par\medskip
          }
        \tl_if_blank:VTF \l__pxtfg_licence_text_tl
          {
            \tl_set:Nx \l_tmpa_tl
              {\tl_upper_case:n {\tl_to_str:V \l__pxtfg_licence_tl}}
            \regex_replace_all:nnN {\s+} {} \l_tmpa_tl
            \exp_args:NV \str_case:nnF \l_tmpa_tl
              {
                {CCBY-NC-SA3.0}
                  {\__pxtfg_cc_notice:nn {3.0}
                    {https://creativecommons.org/licenses/by-nc-sa/3.0/es/}}
                {CCBY-NC-SA4.0}
                  {\__pxtfg_cc_notice:nn {4.0}
                    {https://creativecommons.org/licenses/by-nc-sa/4.0/}}
              }
              {\noindent\tl_use:N \l__pxtfg_licence_tl\par}
          }
          {\noindent\tl_use:N \l__pxtfg_licence_text_tl\par}
      }
  }

% Explicit \pxTFGlicencepage is independent of automatic work-card layout.
\cs_new_protected:Npn \__pxtfg_licence_page:
  {
    \tl_if_blank:VF \l__pxtfg_licence_tl
      {
        \clearpage
        \thispagestyle{empty}
        \group_begin:
          \parindent=0pt
          \vspace*{\fill}
          \begin{minipage}{.85\linewidth}
            \__pxtfg_licence_content:
          \end{minipage}
          \par\vspace*{\fill}
        \group_end:
        \clearpage
      }
  }

% ================================================================
% Abstracts
% ================================================================
\cs_new_protected:Npn \__pxtfg_abstract:nn #1#2
  {\prop_gput:Nnn \g__pxtfg_abstracts_prop {#1}{#2}}

% Chapter headings in MEM, section headings in PR. Both use the TOC.
\cs_new_protected:Npn \__pxtfg_heading:n #1
  {
    \cs_if_exist:NTF \chapter
      {
        \chapter*{#1}
        \cs_if_exist:NT \phantomsection {\phantomsection}
        \addcontentsline{toc}{chapter}{#1}
      }
      {
        \section*{#1}
        \cs_if_exist:NT \phantomsection {\phantomsection}
        \addcontentsline{toc}{section}{#1}
      }
  }

\cs_new_protected:Npn \__pxtfg_print_abstract:n #1
  {
    \prop_get:NnNT \g__pxtfg_abstracts_prop {#1} \l_tmpa_tl
      {
        \__pxtfg_heading:n
          {\str_case:nn {#1}
            {{CAT}{\pxDEFtfgabstractcat}{ES}{\pxDEFtfgabstractes}{EN}{\pxDEFtfgabstracten}}}
        \tl_use:N \l_tmpa_tl\par
      }
  }

\cs_new_protected:Npn \__pxtfg_abstracts:
  {
    \__pxtfg_print_abstract:n {CAT}
    \__pxtfg_print_abstract:n {ES}
    \__pxtfg_print_abstract:n {EN}
  }

\cs_new_protected:Npn \__pxtfg_cat_summary:
  {
    \bool_if:NT \l__pxtfg_abstracts_bool
      {
        \prop_get:NnNT \g__pxtfg_abstracts_prop {CAT} \l_tmpa_tl
          {
            \par\medskip
            \noindent\textbf{\pxDEFtfgsummary}\par\smallskip
            \tl_use:N \l_tmpa_tl\par
          }
      }
  }

\cs_new_protected:Npn \__pxtfg_other_abstracts:
  {
    \bool_if:NT \l__pxtfg_abstracts_bool
      {
        \prop_if_in:NnT \g__pxtfg_abstracts_prop {ES}
          {\clearpage\__pxtfg_print_abstract:n {ES}}
        \prop_if_in:NnT \g__pxtfg_abstracts_prop {EN}
          {\clearpage\__pxtfg_print_abstract:n {EN}}
      }
  }

% ================================================================
% TFG work card: one dedicated page, with the licence at the bottom
% ================================================================
\cs_new:Npn \__pxtfg_row:nn #1#2
  {\textbf{#1} & #2 \\[0.65ex]}

\cs_new_protected:Npn \__pxtfg_card:
  {
    \bool_if:NF \g__pxtfg_card_printed_bool
      {
        \bool_gset_true:N \g__pxtfg_card_printed_bool
        \section*{\pxDEFtfgworkcard}
        \noindent
        \begin{tabular}{@{}p{.29\linewidth}p{.65\linewidth}@{}}
          \__pxtfg_row:nn {\pxDEFtfgtitle}    {\l__pxdata_subj_name_tl}
          \__pxtfg_row:nn {\pxDEFtfgstudent}  {\l__pxdata_student_tl}
          \__pxtfg_row:nn {\pxDEFtfgdegree}   {\l__pxdata_course_tl}
          \__pxtfg_row:nn {\pxDEFtfgactivity} {\l__pxdata_activity_tl}
          \__pxtfg_row:nn {\pxDEFtfgtutor}    {\l__pxtfg_tutor_tl}
          \__pxtfg_row:nn {\pxDEFtfgpra}      {\l__pxtfg_pra_tl}
          \__pxtfg_row:nn {\pxDEFtfgarea}     {\l__pxtfg_area_tl}
          \__pxtfg_row:nn {\pxDEFtfgcredits}  {\l__pxtfg_credits_tl}
          \__pxtfg_row:nn {\pxDEFtfglanguage}{\pxDEFtfglanguagevalue}
          \__pxtfg_row:nn {\pxDEFtfgkeywords} {\l__pxtfg_keywords_tl}
          \__pxtfg_row:nn {\pxDEFtfgdate}     {\l__px_date_tl}
        \end{tabular}
        \par
        \__pxtfg_cat_summary:
        \vspace{\fill}
        \noindent\__pxtfg_licence_content:
        \clearpage
      }
  }

% Optional cover metadata, without empty rows for missing values.
\cs_new_protected:Npn \__pxtfg_cover_row:nN #1#2
  {\tl_if_blank:VF #2 {\textbf{#1:}~\tl_use:N #2\par}}

\cs_new_protected:Npn \__pxtfg_cover_details:
  {
    \__pxtfg_cover_row:nN {\pxDEFtfgconsultant} \l__pxtfg_tutor_tl
    \__pxtfg_cover_row:nN {\pxDEFtfgpra} \l__pxtfg_pra_tl
    \__pxtfg_cover_row:nN {\pxDEFtfgarea} \l__pxtfg_area_tl
  }
