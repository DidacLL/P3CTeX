\cs_new_protected:Npn \__pxtfg_licence_artwork:
  {
    % A recognised CC code selects the bundled PNG; custom licences do not.
    \tl_set:Nx \l_tmpa_tl
      {\tl_lower_case:n {\tl_to_str:V \l__pxtfg_licence_tl}}
    \regex_replace_all:nnN {[^a-z]} {} \l_tmpa_tl
    \str_case:enT {\tl_use:N \l_tmpa_tl}
      {{ccby}{}{ccbync}{}{ccbyncnd}{}{ccbyncsa}{}{ccbynd}{}{ccbysa}{}}
      {
        \clist_map_inline:nn {licenc/,../code/licenc/,tex/code/licenc/}
          {
            \tl_set:Nx \l_tmpb_tl {##1\l_tmpa_tl.png}
            \exp_args:NV \file_if_exist:nT \l_tmpb_tl
              {
                \includegraphics[height=1cm,keepaspectratio]{\l_tmpb_tl}
                \par\medskip
                \clist_map_break:
              }
          }
      }
  }

\cs_new_protected:Npn \__pxtfg_abstract:nn #1#2
  {\prop_gput:Nnn \g__pxtfg_abstracts_prop {#1}{#2}}

% Chapter headings in MEM, section headings in PR. Both use the TOC.
\cs_new_protected:Npn \__pxtfg_heading:n #1
  {
    \cs_if_exist:NTF \chapter
      {
        \chapter*{#1}
        \cs_if_exist:NT \phantomsection {\phantomsection}
        \addcontentsline{toc}{chapter}{#1}
      }
      {
        \section*{#1}
        \cs_if_exist:NT \phantomsection {\phantomsection}
        \addcontentsline{toc}{section}{#1}
      }
  }

\cs_new_protected:Npn \__pxtfg_print_abstract:n #1

            {{CAT}{\pxDEFtfgabstractcat}{ES}{\pxDEFtfgabstractes}{EN}{\pxDEFtfgabstracten}}}
        \tl_use:N \l_tmpa_tl\par
      }
  }

\cs_new_protected:Npn \__pxtfg_other_abstracts:
  {
    \bool_if:NT \l__pxtfg_abstracts_bool
      {
        % The selected-language abstract is already in the local work card.
        \str_if_eq:eeF {\languagename} {spanish}
          {
            \prop_if_in:NnT \g__pxtfg_abstracts_prop {ES}
              {\clearpage\__pxtfg_print_abstract:n {ES}}
          }
        % Do not repeat the English abstract after the TOC when an English
        % information card has already printed it before the TOC.
        \str_if_eq:eeF {\languagename} {english}
          {
            \bool_if:NTF \l__pxtfg_card_bool
              {
                \tl_if_blank:VT \l__pxtfg_title_en_tl
                  {
                    \prop_if_in:NnT \g__pxtfg_abstracts_prop {EN}
                      {\clearpage\__pxtfg_print_abstract:n {EN}}
                  }
              }
              {
                \prop_if_in:NnT \g__pxtfg_abstracts_prop {EN}
                  {\clearpage\__pxtfg_print_abstract:n {EN}}
              }
          }
      }
  }

\cs_new_protected:Npn \__pxtfg_card:
  {
    \bool_if:NF \g__pxtfg_card_printed_bool
      {
        \bool_gset_true:N \g__pxtfg_card_printed_bool
        \tl_if_blank:VTF \l__pxtfg_title_en_tl
          {
            % Untranslated legacy TFGs keep the original single-card page.
            \__pxtfg_local_card:
            \vspace{\fill}
            \noindent\__pxtfg_licence_content:
          }
          {
            \str_if_eq:eeTF {\languagename} {english}
              {
                \__pxtfg_local_card:
                \vspace{\fill}
                \noindent\__pxtfg_licence_content:
              }
              {
                \vbox_set:Nn \l__pxtfg_local_card_box
                  {
                    \hsize=\linewidth
                    \linewidth=\hsize
                    \columnwidth=\hsize
                    \parindent=0pt
                    \tabcolsep=2pt
                    \footnotesize
                    \__pxtfg_local_card:
                  }
                \vbox_set:Nn \l__pxtfg_english_card_box
                  {
                    \hsize=\linewidth
                    \linewidth=\hsize
                    \columnwidth=\hsize
                    \parindent=0pt
                    \tabcolsep=2pt
                    \footnotesize
                    \__pxtfg_english_card:
                  }
                \vbox_set:Nn \l__pxtfg_licence_box
                  {
                    \hsize=\linewidth
                    \parindent=0pt
                    \__pxtfg_licence_content:
                  }
                \dim_set:Nn \l__pxtfg_cards_height_dim
                  {
                    \textheight
                    - \box_ht:N \l__pxtfg_licence_box
                    - \box_dp:N \l__pxtfg_licence_box
                    - 2\baselineskip
                  }
                \dim_set:Nn \l__pxtfg_total_cards_height_dim
                  {
                    \box_ht:N \l__pxtfg_local_card_box
                    + \box_dp:N \l__pxtfg_local_card_box
                    + \box_ht:N \l__pxtfg_english_card_box
                    + \box_dp:N \l__pxtfg_english_card_box
                  }
                \fp_set:Nn \l__pxtfg_card_scale_fp
                  {
                    min
                      (
                        1,
                        \dim_to_fp:n {\l__pxtfg_cards_height_dim}
                        / \dim_to_fp:n {\l__pxtfg_total_cards_height_dim}
                      )
                  }
                \noindent
                \hbox_to_wd:nn {\linewidth}
                  {
                    \hfil
                    \scalebox{\fp_to_decimal:N \l__pxtfg_card_scale_fp}
                      {\box_use:N \l__pxtfg_local_card_box}
                    \hfil
                  }
                \par\smallskip
                \noindent
                \hbox_to_wd:nn {\linewidth}
                  {
                    \hfil
                    \scalebox{\fp_to_decimal:N \l__pxtfg_card_scale_fp}
                      {\box_use:N \l__pxtfg_english_card_box}
                    \hfil
                  }
                \par\vspace{\fill}
                \noindent\box_use:N \l__pxtfg_licence_box
              }
          }
        \clearpage
      }
  }
