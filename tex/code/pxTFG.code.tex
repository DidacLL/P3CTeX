% Manual licence page: reuse bundled CC artwork for recognised codes.
% Other licence descriptions are printed without an assumed logo or legal text.

  {
    \tl_set:Nx \l_tmpa_tl {\tl_lower_case:n {\tl_to_str:V \l__pxtfg_licence_tl}}
    \regex_replace_all:nnN {[^a-z]} {} \l_tmpa_tl
    \str_case:enT {\tl_use:N \l_tmpa_tl}
      {{ccby}{}{ccbync}{}{ccbyncnd}{}{ccbyncsa}{}{ccbynd}{}{ccbysa}{}}
      {
        \clist_map_inline:nn {licenc/,../code/licenc/,tex/code/licenc/}
          {
            \tl_set:Nx \l_tmpb_tl {##1\l_tmpa_tl.png}
            \exp_args:NV \file_if_exist:nT \l_tmpb_tl
              {\includegraphics[height=1cm,keepaspectratio]{\l_tmpb_tl}\par\medskip
               \clist_map_break:}
          }
      }
  }

\cs_new_protected:Npn \__pxtfg_licence_on_card:
  {
    \tl_if_blank:VF \l__pxtfg_licence_tl
      {
        \par\medskip
        \noindent\textbf{\__pxtfg_licence_heading:}:
        \space\tl_use:N \l__pxtfg_licence_tl\par
        \tl_if_blank:VTF \l__pxtfg_licence_text_tl
          {
            % Normalize spaces before comparing codes: under expl3,
            % unescaped spaces in case-list tokens are ignored at definition.
            \tl_set:Nx \l_tmpa_tl
              {\tl_upper_case:n {\tl_to_str:V \l__pxtfg_licence_tl}}
            \regex_replace_all:nnN {\s+} {} \l_tmpa_tl
            \exp_args:NV \str_case:nn \l_tmpa_tl
              {
                {CCBY-NC-SA3.0}{\__pxtfg_cc_terms:n {3.0}}
                {CCBY-NC-SA4.0}{\__pxtfg_cc_terms:n {4.0}}
              }
          }
          {\tl_use:N \l__pxtfg_licence_text_tl\par}
      }
  }
