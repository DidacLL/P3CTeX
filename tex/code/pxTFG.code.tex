% Manual licence page: reuse bundled CC artwork for recognised codes.
% Other licence descriptions are printed without an assumed logo or legal text.
\cs_new_protected:Npn \__pxtfg_licence_artwork:
  {
    \tl_set:Nx \l_tmpa_tl {\tl_lower_case:n {\tl_to_str:V \l__pxtfg_licence_tl}}
    \regex_replace_all:nnN {[^a-z]} {} \l_tmpa_tl
    \str_case:enT {\tl_use:N \l_tmpa_tl}
      {{ccby}{}{ccbync}{}{ccbyncnd}{}{ccbyncsa}{}{ccbynd}{}{ccbysa}{}}
      {
        \clist_map_inline:nn {licenc/,../code/licenc/,tex/code/licenc/}
          {
            \tl_set:Nx \l_tmpb_tl {##1\l_tmpa_tl.png}
            \exp_args:NV \file_if_exist:nT \l_tmpb_tl
              {\includegraphics[height=1cm,keepaspectratio]{\l_tmpb_tl}\par\medskip
               \clist_map_break:}
          }
      }
  }
