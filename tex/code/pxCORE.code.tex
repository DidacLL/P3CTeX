%% pxCORE package keys.
%% Configure at load with \usepackage[<keys>]{pxCORE}; P3CTeX also forwards its unrecognised class option here.
%% PRP/UML/GNT/CMP/SRC/TAB/LST/ORL select modules; UMLconfig/SRCconfig/CMPconfig forward option strings to those packages.
%% default enables the standard PRP+UML+SRC+TAB+LST bundle.

{
	debug		.bool_set:N 	= \l_pxcore_debug_bool,
		debug		.initial:n	= false,
		debug		.default:n	= true,
	print-doc	.bool_set:N 	= \l_pxcore_pxdoc_use_bool,

		PRPconfig	.initial:n	= {},
	GNTconfig	.tl_set_x:N 	= \l_pxcore_gnt_config_tl,
		GNTconfig	.initial:n	= {},
	GNT			.bool_set:N	=	\l_pxcore_gnt_use_bool,
		GNT			.initial:n	=	false,
		GNT			.default:n	=	true,
