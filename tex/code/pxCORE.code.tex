%% ===== pxCORE KEYS DEFINITION ====
%% =================================
%% pxCORE package keys.
%% Configure at load with \usepackage[<keys>]{pxCORE}; P3CTeX also forwards its unrecognised class option here.
%% PRP/UML/PM/DIAG/SRC/TAB/LST/ORL select modules. UMLconfig, SRCconfig,
%% diagrams-config, TABconfig and ORLconfig forward token-preserving package options.
%% TFGconfig applies TFG metadata through pxTFGsetup.
%% default enables the standard PRP+UML+SRC+TAB+LST bundle.
\keys_define:nn { pxCORE }
{
    sources .meta:n = {SRC=#1,LST=#1},
    sources .default:n = true,
    annexes .bool_set:N = \l_pxcore_ann_use_bool,
    annexes .initial:n = false,
    annexes .default:n = true,
    ANN .meta:n = {annexes=#1},
    ANN .default:n = true,
    glossaries .bool_set:N = \l_pxcore_gls_use_bool,
    glossaries .initial:n = false,
    glossaries .default:n = true,
    GLS .meta:n = {glossaries=#1},
    GLS .default:n = true,
    calculations .bool_set:N = \l_pxcore_calc_use_bool,
    calculations .initial:n = false,
    calculations .default:n = true,
    CALC .meta:n = {calculations=#1},
    CALC .default:n = true,

    DIAG .meta:n = {diagrams=#1},
    DIAG .default:n = true,

    code .meta:n = {LST=#1},
    code .default:n = true,
    images .meta:n = {SRC=#1},
    images .default:n = true,
    tables .meta:n = {TAB=#1},
    tables .default:n = true,
    project-management .bool_set:N = \l_pxcore_pm_use_bool,
    project-management .initial:n = false,
    project-management .default:n = true,
    PM .meta:n = {project-management=#1},
    PM .default:n = true,
	debug		.bool_set:N 	= \l_pxcore_debug_bool,
		debug		.initial:n	= false,
		debug		.default:n	= true,
	print-doc	.bool_set:N 	= \l_pxcore_pxdoc_use_bool,
		print-doc	.initial:n	= false,
		print-doc	.default:n	= true,
	PRP			.bool_set:N 	= \l_pxcore_prp_use_bool,
		PRP			.initial:n	= false,
		PRP			.default:n	= true,
	diagrams			.bool_set:N	=	\l_pxcore_diag_use_bool,
		diagrams			.initial:n	=	false,
		diagrams			.default:n	=	true,
	diagrams-config	.tl_set:N 	= \l_pxcore_diag_config_tl,
		diagrams-config	.initial:n	= {},
	UML			.bool_set:N 	= \l_pxcore_uml_use_bool,
		UML			.initial:n	= false,
		UML			.default:n	= true,
	UMLconfig	.tl_set_x:N 	= \l_pxcore_uml_config_tl,
		UMLconfig	.initial:n	= {},
	SRC			.bool_set:N 	= \l_pxcore_src_use_bool,
		SRC			.initial:n	= false,
		SRC			.default:n	= true,
	SRCconfig	.tl_set_x:N 	= \l_pxcore_src_config_tl,
		SRCconfig	.initial:n	= {},
	TAB			.bool_set:N 	= \l_pxcore_tab_use_bool,
		TAB			.initial:n	= false,
		TAB			.default:n	= true,
	TABconfig	.tl_set:N 	= \l_pxcore_tab_config_tl,
		TABconfig	.initial:n	= {},
	LST			.bool_set:N 	= \l_pxcore_lst_use_bool,
		LST			.initial:n	= false,
		LST			.default:n	= true,
	ORL			.bool_set:N = \l_pxcore_orl_use_bool,
		ORL         .initial:n  = false,
		ORL         .default:n  = true,
	ORLconfig	.tl_set:N 	= \l_pxcore_orl_config_tl,
		ORLconfig	.initial:n	= {},
	TFG			.bool_set:N	= \l_pxcore_tfg_use_bool,
		TFG			.initial:n	= false,
		TFG			.default:n	= true,
	MEM			.bool_set:N	= \l_pxcore_mem_use_bool,
		MEM			.initial:n	= false,
		MEM			.default:n	= true,
	TFGconfig	.tl_set:N 	= \l_pxcore_tfg_config_tl,
		TFGconfig	.initial:n	= {},
	default		.meta:n	=	{
		PRP		=	true,
		UML		=	true,
		SRC 	=	true,
		TAB 	=	true,
		LST 	=	true
	},
	SYNTAX		.bool_set:N 	= \l_pxcore_syntax_use_bool,
		SYNTAX		.initial:n	= false,
		SYNTAX		.default:n	= true,
	unknown		.code:n ={}
}
\ProcessKeysOptions{ pxCORE }
%% =================================
%% Module-selection flags are consumed here during package loading.
%% Changing them later with \pxCOREsetup changes state only; it does not re-run these \RequirePackage branches.
%% ===== LOAD SELECTED MODULES =====
%% =================================
\bool_if:NT \l_pxcore_uml_use_bool
	{
		\tl_if_blank:VTF \l_pxcore_uml_config_tl
			{ \RequirePackage{pxUML} }
			{ \RequirePackage[\l_pxcore_uml_config_tl]{pxUML} }
	}
\bool_if:NT \l_pxcore_prp_use_bool { \RequirePackage{pxPRP} }
\bool_if:NT \l_pxcore_src_use_bool {
	\tl_if_blank:VTF \l_pxcore_src_config_tl
	{ \RequirePackage{pxSRC} }
	{ \RequirePackage[\l_pxcore_src_config_tl]{pxSRC} }
}
\bool_if:NT \l_pxcore_tab_use_bool
  {\tl_if_blank:VTF \l_pxcore_tab_config_tl
    {\RequirePackage{pxTAB}}
    {\RequirePackage[\l_pxcore_tab_config_tl]{pxTAB}}}
\bool_if:NT \l_pxcore_lst_use_bool { \RequirePackage{pxLST} }
\bool_if:NT \l_pxcore_orl_use_bool
  {\tl_if_blank:VTF \l_pxcore_orl_config_tl
    {\RequirePackage{pxORL}}
    {\RequirePackage[\l_pxcore_orl_config_tl]{pxORL}}}
\bool_if:NT \l_pxcore_diag_use_bool
	{
		\tl_if_blank:VTF \l_pxcore_diag_config_tl
			{ \RequirePackage{pxDIAG} }
			{ \RequirePackage[\l_pxcore_diag_config_tl]{pxDIAG} }
	}
\bool_if:NT \l_pxcore_mem_use_bool { \RequirePackage{pxMEM} }
\bool_if:NT \l_pxcore_tfg_use_bool
  {\RequirePackage{pxTFG}
   \tl_if_blank:VF \l_pxcore_tfg_config_tl
     {\exp_args:NV \pxTFGsetup \l_pxcore_tfg_config_tl}}

\bool_if:NT \l_pxcore_calc_use_bool { \RequirePackage{pxCALC} }
\bool_if:NT \l_pxcore_pm_use_bool { \RequirePackage{pxPM} }
\bool_if:NT \l_pxcore_ann_use_bool { \RequirePackage{pxANN} }
\bool_if:NT \l_pxcore_gls_use_bool { \RequirePackage{pxGLS} }
