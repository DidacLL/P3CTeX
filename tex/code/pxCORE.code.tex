%% ===== pxCORE KEYS DEFINITION ====
%% =================================
%% pxCORE package keys.
%% Configure at load with \usepackage[<keys>]{pxCORE}; P3CTeX also forwards its unrecognised class option here.
%% PRP/UML/SRC/TAB/LST/ORL select modules; DIA is opt-in. Config keys forward options.
%% default enables the standard PRP+UML+SRC+TAB+LST bundle.
\keys_define:nn { pxCORE }
{
	debug		.bool_set:N 	= \l_pxcore_debug_bool,
		debug		.initial:n	= false,
		debug		.default:n	= true,
	print-doc	.bool_set:N 	= \l_pxcore_pxdoc_use_bool,
		print-doc	.initial:n	= false,
		print-doc	.default:n	= true,
	PRP			.bool_set:N 	= \l_pxcore_prp_use_bool,
		PRP			.initial:n	= false,
		PRP			.default:n	= true,
	UML			.bool_set:N 	= \l_pxcore_uml_use_bool,
		UML			.initial:n	= false,
		UML			.default:n	= true,
	UMLconfig	.tl_set:N 	= \l_pxcore_uml_config_tl,
		UMLconfig	.initial:n	= {},
	SRC			.bool_set:N 	= \l_pxcore_src_use_bool,
		SRC			.initial:n	= false,
		SRC			.default:n	= true,
	SRCconfig	.tl_set:N 	= \l_pxcore_src_config_tl,
		SRCconfig	.initial:n	= {},
	TAB			.bool_set:N 	= \l_pxcore_tab_use_bool,
		TAB			.initial:n	= false,
		TAB			.default:n	= true,
	LST			.bool_set:N 	= \l_pxcore_lst_use_bool,
		LST			.initial:n	= false,
		LST			.default:n	= true,
	ORL			.bool_set:N = \l_pxcore_orl_use_bool,
		ORL         .initial:n  = false,
		ORL         .default:n  = true,
	default		.meta:n	=	{
		PRP		=	true,
		UML		=	true,
		SRC 	=	true,
		TAB 	=	true,
		LST 	=	true
	},
	TFG         .bool_set:N = \l_pxcore_tfg_use_bool,
        TFG         .initial:n = false,
        TFG         .default:n = true,
    MEM         .bool_set:N = \l_pxcore_mem_use_bool,
        MEM         .initial:n = false,
        MEM         .default:n = true,
	CALC       .bool_set:N = \l_pxcore_calc_use_bool,
		CALC       .initial:n = false,
		CALC       .default:n = true,
	calculations .meta:n = { CALC = #1 },
		calculations .default:n = true,
	DIA         .bool_set:N = \l_pxcore_dia_use_bool,
        DIA         .initial:n = false,
        DIA         .default:n = true,
    diagrams    .meta:n = { DIA = #1 },
        diagrams    .default:n = true,
    diagrams-config .tl_set:N = \l_pxcore_dia_config_tl,
        diagrams-config .initial:n = {},
    PM          .bool_set:N = \l_pxcore_pm_use_bool,
        PM          .initial:n = false,
        PM          .default:n = true,
    project-management .meta:n = { PM = #1 },
        project-management .default:n = true,
    ANX         .bool_set:N = \l_pxcore_anx_use_bool,
        ANX         .initial:n = false,
        ANX         .default:n = true,
    annexes .meta:n = {ANX = #1},
        annexes .default:n = true,
    GLS         .bool_set:N = \l_pxcore_gls_use_bool,
        GLS         .initial:n = false,
        GLS         .default:n = true,
    glossaries .meta:n = {GLS = #1},
        glossaries .default:n = true,
    AI          .bool_set:N = \l_pxcore_ai_use_bool,
        AI          .initial:n = false,
        AI          .default:n = true,
    SYNTAX		.bool_set:N 	= \l_pxcore_syntax_use_bool,
		SYNTAX		.initial:n	= false,
		SYNTAX		.default:n	= true,
	unknown		.code:n ={}
}
\ProcessKeysPackageOptions{ pxCORE }
