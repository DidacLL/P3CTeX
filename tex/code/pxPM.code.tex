% pgfcalendar supplies date arithmetic, including month/year boundaries.
\cs_new_protected:Npn \__pxpm_gantt_parse_date:nN #1#2
  {
    \seq_set_split:Nnn \l__pxpm_gantt_date_seq {/} {#1}
    \int_compare:nNnF {\seq_count:N \l__pxpm_gantt_date_seq} = {3}
      {\PackageError{pxPM}{Expected a date in dd/mm/yyyy format}{Correct the date.}}
    \tl_set:Nx \l__pxpm_gantt_iso_tl
      {
        \seq_item:Nn \l__pxpm_gantt_date_seq {3}
        -\seq_item:Nn \l__pxpm_gantt_date_seq {2}
        -\seq_item:Nn \l__pxpm_gantt_date_seq {1}
      }
    \exp_args:NV \pgfcalendardatetojulian \l__pxpm_gantt_iso_tl #2
  }
\cs_new_protected:Npn \__pxpm_gantt_format_date:nN #1#2
  {
    \pgfcalendarjuliantodate{#1}
      {\l__pxpm_gantt_year_tl}{\l__pxpm_gantt_month_tl}{\l__pxpm_gantt_day_tl}
    \tl_set:Nx #2
      {\l__pxpm_gantt_day_tl/\l__pxpm_gantt_month_tl/\l__pxpm_gantt_year_tl}
  }

\cs_new_protected:Npn \__pxpm_gantt_begin:n #1
  {
    \seq_gclear:N \g__pxpm_gantt_items_seq
    \tl_clear:N \l__pxpm_gantt_start_date_tl
    \tl_clear:N \l__pxpm_gantt_end_date_tl
    \bool_set_false:N \l__pxpm_gantt_fixed_label_width_bool
    \bool_set_false:N \l__pxpm_gantt_fixed_day_width_bool
    \bool_set_false:N \l__pxpm_gantt_rotate_bool
    \int_set:Nn \l__pxpm_gantt_days_int {28}
    \dim_zero:N \l__pxpm_gantt_label_dim
    \dim_set:Nn \l__pxpm_gantt_label_max_dim {.24\linewidth}
    \keys_set:nn {pxPM}{#1}
    \tl_if_blank:VT \l__pxpm_gantt_start_date_tl
      {\PackageError{pxPM}{Missing required start date}{Set start=dd/mm/yyyy.}}
    \tl_if_blank:VT \l__pxpm_gantt_end_date_tl
      {\PackageError{pxPM}{Missing required end date}{Set end=dd/mm/yyyy.}}
    \int_compare:nNnF {\l__pxpm_gantt_days_int} > {0}
      {\PackageError{pxPM}{days-per-block must be positive}{Use a positive integer.}}
    \bool_if:NT \l__pxpm_gantt_fixed_day_width_bool
      {
        \dim_compare:nNnF {\l__pxpm_gantt_requested_day_dim} > {0pt}
          {\PackageError{pxPM}{day-width must be positive}{Specify a positive dimension.}}
      }
    \exp_args:NV \__pxpm_gantt_parse_date:nN \l__pxpm_gantt_start_date_tl \l__pxpm_gantt_start_int
    \exp_args:NV \__pxpm_gantt_parse_date:nN \l__pxpm_gantt_end_date_tl \l__pxpm_gantt_end_int
    \int_compare:nNnT {\l__pxpm_gantt_start_int} > {\l__pxpm_gantt_end_int}
      {\PackageError{pxPM}{Chart end date precedes its start}{Correct start and end.}}
  }

        \seq_gput_right:Nn \g__pxpm_gantt_items_seq

  }

\cs_new_protected:Npn \__pxpm_gantt_bar:nnn #1#2#3

    \__pxpm_gantt_parse_date:nN {#2} \l__pxpm_gantt_item_start_int

    \bool_if:nT
      {

        \int_compare_p:n {\l__pxpm_gantt_item_start_int <= \l__pxpm_gantt_window_end_int}
      }
      {
        \__pxpm_gantt_format_date:nN
          {\int_max:nn {\l__pxpm_gantt_item_start_int}{\l__pxpm_gantt_window_start_int}}

        \__pxpm_gantt_format_date:nN
          {\int_min:nn {\l__pxpm_gantt_item_end_int}{\l__pxpm_gantt_window_end_int}}
          \l__pxpm_gantt_bar_end_date_tl
        \bool_if:NF \l__pxpm_gantt_first_bool {\ganttnewline}

              bar/.append~style={fill=primaryColor!65!secondaryColor,draw=primaryColor}
            ]{#1}{\tl_use:N \l__pxpm_gantt_bar_start_date_tl}

\cs_new_protected:Npn \__pxpm_gantt_render:
  {
    \dim_set:Nn \l__pxpm_gantt_chart_width_dim
      {\bool_if:NTF \l__pxpm_gantt_rotate_bool {\textheight} {\linewidth}}
    \bool_if:NT \l__pxpm_gantt_fixed_day_width_bool
      {
        \int_set:Nn \l__pxpm_gantt_fit_days_int
          {\fp_to_int:n
            {floor(
              \dim_to_fp:n {\l__pxpm_gantt_chart_width_dim - \l__pxpm_gantt_label_dim - 2\labelsep}
              / \dim_to_fp:n {\l__pxpm_gantt_requested_day_dim})}}
        \int_compare:nNnT {\l__pxpm_gantt_fit_days_int} < {1}
          {\PackageError{pxPM}{day-width cannot fit alongside labels}
             {Reduce day-width or label-width.}}
        \int_set:Nn \l__pxpm_gantt_days_int
          {\int_min:nn {\l__pxpm_gantt_days_int}{\int_max:nn {1}{\l__pxpm_gantt_fit_days_int}}}
      }
    \int_set:Nn \l__pxpm_gantt_window_count_int
      {\int_div_truncate:nn
        {\l__pxpm_gantt_end_int-\l__pxpm_gantt_start_int+\l__pxpm_gantt_days_int}
        {\l__pxpm_gantt_days_int}}
    \int_set:Nn \l__pxpm_gantt_days_int
      {\int_div_truncate:nn

            {\l__pxpm_gantt_end_int}}

      }
