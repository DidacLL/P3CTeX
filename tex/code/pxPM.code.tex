% pgfcalendar supplies date arithmetic, including month/year boundaries.
\cs_new_protected:Npn \__pxpm_gantt_parse_date:nN #1#2
  {
    \exp_args:NNne \seq_set_split:Nnn \l__pxpm_gantt_date_seq {/} {#1}
    \int_compare:nNnF {\seq_count:N \l__pxpm_gantt_date_seq} = {3}
      {\PackageError{pxPM}{Expected a date in dd/mm/yyyy format}{Correct the date.}}
    \tl_set:Nx \l__pxpm_gantt_iso_tl
      {
        \seq_item:Nn \l__pxpm_gantt_date_seq {3}
        -\seq_item:Nn \l__pxpm_gantt_date_seq {2}
        -\seq_item:Nn \l__pxpm_gantt_date_seq {1}
      }
    \exp_args:NV \pgfcalendardatetojulian \l__pxpm_gantt_iso_tl #2
  }
\cs_new_protected:Npn \__pxpm_gantt_format_date:nN #1#2
  {
    \pgfcalendarjuliantodate{#1}
      {\l__pxpm_gantt_year_tl}{\l__pxpm_gantt_month_tl}{\l__pxpm_gantt_day_tl}
    \tl_set:Nx #2
      {\l__pxpm_gantt_day_tl/\l__pxpm_gantt_month_tl/\l__pxpm_gantt_year_tl}
  }
