%% \__pxsrc_label_from_string:nN {<source text>} <tl>
%% Sanitize a path/name into a label-safe string and prepend the configured label-prefix.
%% #1 - source text.  #2 - destination token list.
\cs_new_protected:Npn \__pxsrc_label_from_string:nN #1 #2
	{
		\tl_set:Nn #2 { #1 }
		% Drop a leading ./ which is common in relative paths.
		\regex_replace_once:nnN { \A \./ } { } #2
		% Keep only characters that are safe for \label (csname-based storage).
		\regex_replace_all:nnN { [^-A-Za-z0-9:_.]+ } { - } #2
		\regex_replace_all:nnN { -+ } { - } #2
		\tl_set:Nx #2 { \l__pxsrc_label_prefix_tl \tl_use:N #2 }
	}

%% \__pxsrc_resolve_graphic:nN {<path>} <tl>
%% Resolve an image path as given, then try .png, .pdf and .jpg; leave destination blank if none exists.
%% #1 - requested path/name.  #2 - destination tl for the resolved filename.
\cs_new:Npn \__pxsrc_resolve_graphic:nN #1 #2{
		\tl_clear:N #2
		\file_if_exist:nTF { #1 }{ \tl_set:Nn #2 { #1 } }{
				\file_if_exist:nTF { #1 .png }{ \tl_set:Nn #2 { #1 .png } }{
						\file_if_exist:nTF { #1 .pdf }{ \tl_set:Nn #2 { #1 .pdf } }{
								\file_if_exist:nTF { #1 .jpg }{ \tl_set:Nn #2 { #1 .jpg}}{}
			}
		}
	}
}

	}

    \exp_args:NNV \ior_open:Nn \l__pxsrc_input_ior \l__pxsrc_input_path_tl

      {\int_incr:N \l__pxsrc_input_lines_int

            {\l__pxsrc_input_region_prefix_tl <*\l__pxsrc_input_region_tl>\l__pxsrc_input_region_suffix_tl}

          \str_if_eq:eeT {\tl_trim_spaces:n {##1}}

        {\int_compare_p:n {\l__pxsrc_input_ends_int != 1}}
        {\__pxsrc_input_error:n {A~region~must~have~exactly~one~start~and~one~end~marker.}}}
    \int_compare:nNnT {\l__pxsrc_input_last_int} = {0}
      {\int_set_eq:NN \l__pxsrc_input_last_int \l__pxsrc_input_lines_int}

           {\box_ht:N \l__pxsrc_preview_box+\box_dp:N \l__pxsrc_preview_box}
