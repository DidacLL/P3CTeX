\msg_new:nnn {pxGLS}{type}{Unknown~glossary~type~'#1'.~Define~it~before~printing.}
\prop_new:N \g__pxgls_printed_prop
\tl_new:N \l__pxgls_backend_tl
\tl_new:N \l__pxgls_type_tl
\tl_new:N \l__pxgls_title_tl
\tl_new:N \l__pxgls_terms_title_tl
\tl_new:N \l__pxgls_acronyms_title_tl
\tl_new:N \l__pxgls_options_tl
\bool_new:N \l__pxgls_auto_bool
\bool_new:N \l__pxgls_backend_locked_bool
\bool_new:N \l__pxgls_all_bool
\bool_new:N \l__pxgls_terms_bool
\bool_new:N \l__pxgls_acronyms_bool
\bool_new:N \l__pxgls_toc_bool
\keys_define:nn {pxGLS}
  {
    backend .choices:nn = {noindex,external}{\bool_if:NTF \l__pxgls_backend_locked_bool
      {\tl_if_eq:NNF \l__pxgls_backend_tl \l_keys_choice_tl
        {\PackageError{pxGLS}{Glossary~backend~is~a~package-load~option}{Choose~the~backend~before~defining~entries.}}}
      {\tl_set_eq:NN \l__pxgls_backend_tl \l_keys_choice_tl}},
    backend .initial:n = noindex,
    print .choice:,
    print / manual .code:n = {\bool_set_false:N \l__pxgls_auto_bool},
    print / auto .code:n = {\bool_set_true:N \l__pxgls_auto_bool},
    print .initial:n = auto,
    include .choice:,
    include / used .code:n = {\bool_set_false:N \l__pxgls_all_bool},
    include / all .code:n = {\bool_set_true:N \l__pxgls_all_bool},
    include .initial:n = used,
    terms .bool_set:N = \l__pxgls_terms_bool,
    terms .initial:n = true,
    acronyms .bool_set:N = \l__pxgls_acronyms_bool,
    acronyms .initial:n = true,
    toc .bool_set:N = \l__pxgls_toc_bool,
    toc .initial:n = true,
    type .tl_set:N = \l__pxgls_type_tl,
    type .initial:n = main,
    title .tl_set:N = \l__pxgls_title_tl,
    terms-title .tl_set:N = \l__pxgls_terms_title_tl,
    terms-title .initial:n = \glossaryname,
    acronyms-title .tl_set:N = \l__pxgls_acronyms_title_tl,
    acronyms-title .initial:n = \acronymname,
    options .tl_set:N = \l__pxgls_options_tl
  }
\cs_new_protected:Npn \__pxgls_print:
  {
    \prop_if_in:NVF \g__pxgls_printed_prop \l__pxgls_type_tl
      {
        \exp_args:NV \ifglossaryexists \l__pxgls_type_tl
          {
            \group_begin:
            \bool_if:NT \l__pxgls_all_bool
              {\use:x {\exp_not:N \glsaddall[types={\l__pxgls_type_tl}]}}
            \tl_if_blank:VT \l__pxgls_title_tl
              {\tl_if_eq:NnTF \l__pxgls_type_tl {acronym}
                {\tl_set_eq:NN \l__pxgls_title_tl \l__pxgls_acronyms_title_tl}
                {\tl_set_eq:NN \l__pxgls_title_tl \l__pxgls_terms_title_tl}}
            \bool_if:NTF \l__pxgls_toc_bool {\glstoctrue}{\glstocfalse}
            \tl_if_eq:NnTF \l__pxgls_backend_tl {noindex}
              {\use:x {\exp_not:N \printnoidxglossary[
                type={\l__pxgls_type_tl},title={\exp_not:V \l__pxgls_title_tl},
                \exp_not:V \l__pxgls_options_tl]}}
              {\use:x {\exp_not:N \printglossary[
                type={\l__pxgls_type_tl},title={\exp_not:V \l__pxgls_title_tl},
                \exp_not:V \l__pxgls_options_tl]}}
            \prop_gput:NVn \g__pxgls_printed_prop \l__pxgls_type_tl {true}
            \group_end:
          }{}
      }
  }
%% Batch registration keeps the native glossaries entry model and error checks.
\msg_new:nnn {pxGLS}{entry}{Entry~'#1'~needs~a~braced~field~list.}
\msg_new:nnn {pxGLS}{acronym}{Acronym~'#1'~requires~short~and~long~fields.}
\tl_new:N \l__pxgls_short_tl
\tl_new:N \l__pxgls_long_tl
\tl_new:N \l__pxgls_entry_options_tl
\keys_define:nn {pxGLS/acronym}
  {short .tl_set:N = \l__pxgls_short_tl,
   long .tl_set:N = \l__pxgls_long_tl,
   unknown .code:n = {\tl_put_right:Nx \l__pxgls_entry_options_tl
     {\l_keys_key_str={\exp_not:n {#1}},}}}
\cs_new_protected:Npn \__pxgls_missing_entry:n #1
  {\msg_error:nnn {pxGLS}{entry}{#1}}
\cs_new_protected:Npn \__pxgls_term:nn #1#2
  {\newglossaryentry{#1}{#2}}
\cs_new_protected:Npn \__pxgls_acronym:nn #1#2
  {\group_begin:
   \tl_clear:N \l__pxgls_short_tl
   \tl_clear:N \l__pxgls_long_tl
   \tl_clear:N \l__pxgls_entry_options_tl
   \keys_set:nn {pxGLS/acronym}{#2}
   \bool_lazy_or:nnTF {\tl_if_blank_p:V \l__pxgls_short_tl}
     {\tl_if_blank_p:V \l__pxgls_long_tl}
     {\msg_error:nnn {pxGLS}{acronym}{#1}}
     {\use:x {\exp_not:N \newacronym[\exp_not:V \l__pxgls_entry_options_tl]
       {\exp_not:n {#1}}{\exp_not:V \l__pxgls_short_tl}{\exp_not:V \l__pxgls_long_tl}}}
   \group_end:}
