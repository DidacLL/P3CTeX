%% Space-limited previews of literal files. A preview never executes source TeX.
\ProvidesExplFile{pxSRC.preview.code.tex}{2026/10/31}{0.1}{Measured source previews}
\dim_new:N \l__pxsrc_preview_height_dim
\tl_new:N \l__pxsrc_preview_title_tl
\tl_new:N \l__pxsrc_preview_label_tl
\tl_new:N \l__pxsrc_preview_options_tl
\int_new:N \g__pxsrc_preview_int
\int_new:N \l__pxsrc_preview_first_int
\int_new:N \l__pxsrc_preview_low_int
\int_new:N \l__pxsrc_preview_high_int
\int_new:N \l__pxsrc_preview_mid_int
\box_new:N \l__pxsrc_preview_box
\box_new:N \l__pxsrc_preview_best_box
%% Preview-only keys; remaining options use the existing source input family.
\keys_define:nn {pxSRC/preview}
  {height .dim_set:N = \l__pxsrc_preview_height_dim,
   annex-title .tl_set:N = \l__pxsrc_preview_title_tl,
   annex-label .tl_set:N = \l__pxsrc_preview_label_tl}
\providecommand\pxDEFpreviewname{Preview}
\providecommand\pxDEFfullsource{Full~source~in}
\providecommand\pxDEFpreviewempty{No~complete~source~line~fits.}
\tcbset{pxsrc-preview/.style={
  colback=pxbg!95!,colframe=pxbg!70!,coltext=pxfg,
  colbacktitle=pxbg!60!,coltitle=pxfg,fonttitle=\small\bfseries,
  boxrule=.5pt,arc=.25em,before~skip=0pt,after~skip=0pt}}
%% \__pxsrc_preview_measure:n {<line count>}
%% Measure the complete styled box, including wraps, heading and annex link.
%% Source bounds and styles are local; the accepted box is reused without re-reading.
\cs_new_protected:Npn \__pxsrc_preview_measure:n #1
  {\vbox_set:Nn \l__pxsrc_preview_box
    {\hsize=\linewidth
     \hypersetup{linkcolor=pxfg}
     \noindent\begin{minipage}{\linewidth}
     \begin{tcolorbox}[pxsrc-preview,title={\pxDEFpreviewname\space---\space\l__pxsrc_input_title_tl}]
     \int_compare:nNnTF {#1}>{0}
       {\use:x {\exp_not:N \lstinputlisting[
          style=pxlst,language={\exp_not:V \l__pxsrc_input_language_tl},
          \exp_not:V \l__pxsrc_input_options_tl,
          escapeinside={},escapechar={},mathescape=false,texcl=false,
          firstline=\int_use:N \l__pxsrc_preview_first_int,
          lastline=\int_eval:n {\l__pxsrc_preview_first_int+#1-1},
          firstnumber=\int_use:N \l__pxsrc_preview_first_int]
          {\exp_not:V \l__pxsrc_input_path_tl}}}
       {\pxDEFpreviewempty}
     \tcblower
     \use:x {\exp_not:N \hyperref[\l__pxsrc_preview_label_tl]
       {\exp_not:N \pxDEFfullsource\exp_not:N \space\exp_not:N \appendixname\exp_not:N \space
        \exp_not:N \ref*{\l__pxsrc_preview_label_tl}}}
     \end{tcolorbox}
     \end{minipage}}}
%% \__pxsrc_preview:nn {<options>}{<path>}
%% Validate selection, find the largest complete-line prefix that fits, then queue
%% the whole original file. Preview metadata is applied only once, in the annex.
\cs_new_protected:Npn \__pxsrc_preview:nn #1#2
  {\group_begin:
   \dim_set:Nn \l__pxsrc_preview_height_dim {\textheight}
   \tl_clear:N \l__pxsrc_preview_title_tl
   \tl_clear:N \l__pxsrc_preview_label_tl
   \keys_set_known:nnN {pxSRC/preview}{#1}\l__pxsrc_preview_options_tl
   \keys_set:nn {pxSRC/input}{as=code}
   \exp_args:NnV \keys_set:nn {pxSRC/input}\l__pxsrc_preview_options_tl
   \bool_set_true:N \l__pxsrc_input_valid_bool
   \bool_set_true:N \l__pxsrc_input_fragment_bool
   \tl_if_eq:NnT \l__pxsrc_input_mode_tl {native}
     {\__pxsrc_input_error:n {Previews~support~as=code~or~as=text.}}
   \cs_if_exist:NF \pxAddAnnex
     {\__pxsrc_input_error:n {Previews~require~the~annexes~tool.}}
   \cs_if_exist:NF \lstinputlisting
     {\__pxsrc_input_error:n {Previews~require~the~sources~or~code~tool.}}
   \file_get_full_name:nNTF {#2}\l__pxsrc_input_path_tl
     {\__pxsrc_input_select:
      \bool_if:NT \l__pxsrc_input_valid_bool
        {\int_gincr:N \g__pxsrc_preview_int
         \tl_if_blank:VT \l__pxsrc_preview_label_tl
           {\tl_set:Nx \l__pxsrc_preview_label_tl {pxsrc:full:\int_use:N \g__pxsrc_preview_int}}
         \tl_if_blank:VT \l__pxsrc_input_title_tl
           {\tl_set:Nx \l__pxsrc_input_title_tl {\tl_to_str:n {#2}}}
         \tl_if_blank:VT \l__pxsrc_preview_title_tl
           {\tl_set_eq:NN \l__pxsrc_preview_title_tl \l__pxsrc_input_title_tl}
         \tl_if_eq:NnT \l__pxsrc_input_mode_tl {text}
           {\tl_clear:N \l__pxsrc_input_language_tl}
         \int_set_eq:NN \l__pxsrc_preview_first_int \l__pxsrc_input_first_int
         \int_zero:N \l__pxsrc_preview_low_int
         \int_set:Nn \l__pxsrc_preview_high_int
           {\l__pxsrc_input_last_int-\l__pxsrc_input_first_int+1}
         \__pxsrc_preview_measure:n {0}
         \dim_compare:nNnTF
           {\box_ht:N \l__pxsrc_preview_box+\box_dp:N \l__pxsrc_preview_box}
           > {\l__pxsrc_preview_height_dim}
           {\__pxsrc_input_error:n {Preview~height~cannot~fit~its~heading~and~annex~link.}}
           {\box_set_eq:NN \l__pxsrc_preview_best_box \l__pxsrc_preview_box
            \int_while_do:nn {\l__pxsrc_preview_low_int < \l__pxsrc_preview_high_int}
              {\int_set:Nn \l__pxsrc_preview_mid_int
                 {\int_div_truncate:nn {\l__pxsrc_preview_low_int+\l__pxsrc_preview_high_int+1}{2}}
               \__pxsrc_preview_measure:n {\l__pxsrc_preview_mid_int}
               \dim_compare:nNnTF
                 {\box_ht:N \l__pxsrc_preview_box+\box_dp:N \l__pxsrc_preview_box}
                 > {\l__pxsrc_preview_height_dim}
                 {\int_set:Nn \l__pxsrc_preview_high_int {\l__pxsrc_preview_mid_int-1}}
                 {\int_set_eq:NN \l__pxsrc_preview_low_int \l__pxsrc_preview_mid_int
                  \box_set_eq:NN \l__pxsrc_preview_best_box \l__pxsrc_preview_box}}
            \use:x {\exp_not:N \pxAddAnnex[label={\l__pxsrc_preview_label_tl}]
              {\exp_not:V \l__pxsrc_preview_title_tl}
              {\exp_not:N \pxInput[as=code,\exp_not:V \l__pxsrc_preview_options_tl,
                region={},first-line=1,last-line=0,container=none]
                {\exp_not:V \l__pxsrc_input_path_tl}}}
            \par\nointerlineskip\box_use_drop:N \l__pxsrc_preview_best_box\par}}
     }
     {\__pxsrc_input_error:n {Source~file~'#2'~not~found.}}
   \group_end:}
