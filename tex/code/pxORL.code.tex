%% \pxORL_save_preset:nn {<name>}{<key list>}

%% \pxORL_use_preset:n {<name>}
%% Apply a stored preset through the normal pxORL key parser; warn when unknown.
%% #1 - preset name.
\cs_new_protected:Npn \pxORL_use_preset:n #1
{
	\prop_get:NnNTF \g_pxorl_presets_prop { #1 } \l_tmpa_tl
	{ \keys_set:nV { pxORL } \l_tmpa_tl }
	{ \msg_warning:nnn { pxORL } { unknown-preset } { #1 } }
}

%% #1 - tl variable containing the filename.  #2 - target height.  #3 - maximum width.

			]{\tl_use:N #1}
		}
		{ }
	}

%% \__pxorl_select_columns:
%% Legacy column selector: 2 columns below 5 students, 3 below 10, otherwise 4.
\cs_set_protected:Npn \__pxorl_select_columns:
{
	\int_compare:nNnTF { \l__pxorl_count_int } < { 5 }
	{ \int_set:Nn \l__pxorl_columns_int { 2 } }
	{
		\int_compare:nNnTF { \l__pxorl_count_int } < { 10 }
		{ \int_set:Nn \l__pxorl_columns_int { 3 } }
		{ \int_set:Nn \l__pxorl_columns_int { 4 } }
	}
}
%% \__pxorl_select_layout:
%% Resolve automatic column count and automatic card/photo dimensions from \l__pxorl_count_int.
\cs_new_protected:Npn \__pxorl_select_layout:
{
	\bool_if:NT \l__pxorl_auto_columns_bool
	{
		\int_compare:nNnTF { \l__pxorl_count_int } < { 2 }
		{ \int_set:Nn \l__pxorl_columns_int { 1 } }
		{
			\int_compare:nNnTF { \l__pxorl_count_int } < { 5 }
			{ \int_set:Nn \l__pxorl_columns_int { 2 } }
			{
				\int_compare:nNnTF { \l__pxorl_count_int } < { 10 }
				{ \int_set:Nn \l__pxorl_columns_int { 3 } }
				{
					\int_compare:nNnTF { \l__pxorl_count_int } < { 17 }
					{ \int_set:Nn \l__pxorl_columns_int { 4 } }
					{ \int_set:Nn \l__pxorl_columns_int { 5 } }
				}
			}
		}
	}

	\bool_if:NT \l__pxorl_auto_size_bool
	{
		\int_compare:nNnTF { \l__pxorl_count_int } < { 7 }
		{
			\dim_set:Nn \l__pxorl_photo_width_dim  { 4.05cm }
			\dim_set:Nn \l__pxorl_photo_height_dim { 5.10cm }
			\dim_set:Nn \l__pxorl_card_width_dim   { 4.65cm }
			\dim_set:Nn \l__pxorl_card_sep_dim     { .70cm }
			\dim_set:Nn \l__pxorl_row_sep_dim      { .95cm }
			\tl_set:Nn  \l__pxorl_name_size_tl     { \normalsize }
		}
		{
			\int_compare:nNnTF { \l__pxorl_count_int } < { 13 }
			{
				\dim_set:Nn \l__pxorl_photo_width_dim  { 3.05cm }
				\dim_set:Nn \l__pxorl_photo_height_dim { 3.85cm }
				\dim_set:Nn \l__pxorl_card_width_dim   { 3.55cm }
				\dim_set:Nn \l__pxorl_card_sep_dim     { .48cm }
				\dim_set:Nn \l__pxorl_row_sep_dim      { .85cm }
				\tl_set:Nn  \l__pxorl_name_size_tl     { \small }
			}
			{
				\dim_set:Nn \l__pxorl_photo_width_dim  { 2.45cm }
				\dim_set:Nn \l__pxorl_photo_height_dim { 3.10cm }
				\dim_set:Nn \l__pxorl_card_width_dim   { 2.95cm }
				\dim_set:Nn \l__pxorl_card_sep_dim     { .35cm }
				\dim_set:Nn \l__pxorl_row_sep_dim      { .65cm }
				\tl_set:Nn  \l__pxorl_name_size_tl     { \scriptsize }
			}
		}
	}
}

%% \__pxorl_placeholder:n {<text>}
%% Draw a fixed-size framed placeholder matching the configured photo dimensions.
%% #1 - text displayed below the placeholder question mark.
\cs_set_protected:Npn \__pxorl_placeholder:n #1
{
	% TeX-only fixed-size placeholder. Do not use a TikZ node here:
	% the previous implementation drew the frame correctly but placed the
	% placeholder text off-centre in some engines/viewers.
	\begingroup
		\setlength{\fboxrule}{.45pt}
		\setlength{\fboxsep}{0pt}
		\fcolorbox{\l__pxorl_neutral_color_tl}{\l__pxorl_photo_bg_color_tl}
		{
			\makebox[\dimexpr\l__pxorl_photo_width_dim-2\fboxrule\relax][c]
			{
				\vbox to \dimexpr\l__pxorl_photo_height_dim-2\fboxrule\relax
				{
					\vfil
					\hbox to \dimexpr\l__pxorl_photo_width_dim-2\fboxrule\relax
					{
						\hfil
						\begin{minipage}{\dimexpr\l__pxorl_photo_width_dim-2\fboxrule\relax}
							\centering
							{\Large\bfseries\color{\l__pxorl_neutral_color_tl}?\par}
							\vspace{.35em}
							{\scriptsize\color{\l__pxorl_neutral_color_tl}#1\par}
						\end{minipage}
						\hfil
					}
					\vfil
				}
			}
		}
	\endgroup
}

%% \__pxorl_centered_photo_box:n {<content>}
%% Vertically and horizontally center content inside the configured photo rectangle.
%% #1 - image or empty content.
\cs_new_protected:Npn \__pxorl_centered_photo_box:n #1
{
	\makebox[\l__pxorl_photo_width_dim][c]{%
		\vbox to \l__pxorl_photo_height_dim {%
			\vfil
			\hbox{#1}%
			\vfil
		}%
	}%
}

%% \__pxorl_photo:n {<filename>}
%% Resolve <photos-path><filename>; include it proportionally or emit/leave a placeholder according to print-placeholders.
%% #1 - student photo filename; blank is treated as a missing photo.
\cs_set_protected:Npn \__pxorl_photo:n #1
{
	% \foreach can leave trailing space tokens in the last item of the list
	% when the user formats the argument across lines.  That makes the final
	% file lookup become, for example, `img/orla/nuria.jpg<space>`, so
	% \IfFileExists fails and prints the placeholder.  Normalise the received
	% photo field before testing or including the file.
	\tl_set:Nn \l_tmpa_tl { #1 }
	\tl_trim_spaces:N \l_tmpa_tl
	\tl_set_eq:NN \l_tmpb_tl \l__pxorl_photos_path_tl
	\tl_trim_spaces:N \l_tmpb_tl

	\tl_if_blank:VTF \l_tmpa_tl
	{
		\bool_if:NTF \l__pxorl_print_placeholders_bool
		{ \__pxorl_placeholder:n { Foto~pendent } }
		{ \__pxorl_centered_photo_box:n { } }
	}
	{
		\IfFileExists{\tl_use:N \l_tmpb_tl \tl_use:N \l_tmpa_tl}
		{
			\__pxorl_centered_photo_box:n
			{
				\includegraphics[
				width=\l__pxorl_photo_width_dim,
				height=\l__pxorl_photo_height_dim,
				keepaspectratio
				]{\tl_use:N \l_tmpb_tl \tl_use:N \l_tmpa_tl}
			}
		}
		{
			\bool_if:NTF \l__pxorl_print_placeholders_bool
			{ \__pxorl_placeholder:n { Foto~pendent } }
			{ \__pxorl_centered_photo_box:n { } }
		}
	}
}

%% Main renderer
%% -------------------------------------------------------
%% \pxORL_print:n {<students>}
%% Render one complete orla page from a comma list of {Name}/{photo-file} pairs.
%% #1 - student entries consumed by pgffor; empty photo fields are allowed.
\cs_set_protected:Npn \pxORL_print:n #1
{
	\clist_set:Nn \l__pxorl_students_clist { #1 }
	\int_set:Nn \l__pxorl_count_int
	{ \clist_count:N \l__pxorl_students_clist }

	\__pxorl_select_layout:

	% Do not call \clearpage unconditionally here.
	% At the start of a document it creates the blank first page seen in the test PDF.
	% Only clear when the current page already has material.
	\bool_if:NT \l__pxorl_clearpage_bool
	{
		\ifdim\pagetotal>0pt
			\clearpage
		\fi
	}

	\thispagestyle{empty}
	\setlength{\parindent}{0pt}

	% Render as a real page overlay, but draw the frame with TikZ instead
	% of using \fcolorbox. \fcolorbox around a fixed-height minipage can
	% still contribute depth and push the visible box below the page.
	\null
	\begin{tikzpicture}[remember~picture, overlay]
		% Page-safe background and outer frame.
		\fill[
			\l__pxorl_paper_color_tl
		]
			([xshift=\l__pxorl_page_margin_dim,
			  yshift=-\l__pxorl_page_margin_dim]current~page.north~west)
			rectangle
			([xshift=-\l__pxorl_page_margin_dim,
			  yshift=\dimexpr\l__pxorl_page_margin_dim+\l__pxorl_page_safety_dim\relax]current~page.south~east);

		\draw[
			\l__pxorl_frame_color_tl,
			line~width=.75pt
		]
			([xshift=\l__pxorl_page_margin_dim,
			  yshift=-\l__pxorl_page_margin_dim]current~page.north~west)
			rectangle
			([xshift=-\l__pxorl_page_margin_dim,
			  yshift=\dimexpr\l__pxorl_page_margin_dim+\l__pxorl_page_safety_dim\relax]current~page.south~east);

		% Inner thin frame, equivalent to the earlier double-frame look,
		% but also constrained to the physical page.
		\draw[
			\l__pxorl_neutral_color_tl,
			line~width=.35pt
		]
			([xshift=\dimexpr\l__pxorl_page_margin_dim+4pt\relax,
