% pgfcalendar supplies date arithmetic, including month/year boundaries.
\cs_new_protected:Npn \__pxgnt_parse_date:nN #1#2
  {
    \seq_set_split:Nnn \l__pxgnt_date_seq {/} {#1}
    \int_compare:nNnF {\seq_count:N \l__pxgnt_date_seq} = {3}
      {\PackageError{pxGNT}{Expected a date in dd/mm/yyyy format}{Correct the date.}}
    \tl_set:Nx \l__pxgnt_iso_tl
      {
        \seq_item:Nn \l__pxgnt_date_seq {3}
        -\seq_item:Nn \l__pxgnt_date_seq {2}
        -\seq_item:Nn \l__pxgnt_date_seq {1}
      }
    \exp_args:NV \pgfcalendardatetojulian \l__pxgnt_iso_tl #2
  }
\cs_new_protected:Npn \__pxgnt_format_date:nN #1#2
  {
    \pgfcalendarjuliantodate{#1}
      {\l__pxgnt_year_tl}{\l__pxgnt_month_tl}{\l__pxgnt_day_tl}
    \tl_set:Nx #2
      {\l__pxgnt_day_tl/\l__pxgnt_month_tl/\l__pxgnt_year_tl}
  }

\cs_new_protected:Npn \__pxgnt_begin:n #1
  {
    \seq_gclear:N \g__pxgnt_items_seq
    \bool_set_false:N \l__pxgnt_fixed_label_width_bool
    \bool_set_false:N \l__pxgnt_fixed_day_width_bool
    \bool_set_false:N \l__pxgnt_rotate_bool
    \int_set:Nn \l__pxgnt_days_int {28}
    \dim_zero:N \l__pxgnt_label_dim
    \dim_set:Nn \l__pxgnt_label_max_dim {.24\linewidth}
    \keys_set:nn {pxGNT}{#1}
    \int_compare:nNnF {\l__pxgnt_days_int} > {0}
      {\PackageError{pxGNT}{days-per-block must be positive}{Use a positive integer.}}
    \bool_if:NT \l__pxgnt_fixed_day_width_bool
      {
        \dim_compare:nNnF {\l__pxgnt_requested_day_dim} > {0pt}
          {\PackageError{pxGNT}{day-width must be positive}{Specify a positive dimension.}}
      }
    \exp_args:NV \__pxgnt_parse_date:nN \l__pxgnt_start_date_tl \l__pxgnt_start_int
    \exp_args:NV \__pxgnt_parse_date:nN \l__pxgnt_end_date_tl \l__pxgnt_end_int
    \int_compare:nNnT {\l__pxgnt_start_int} > {\l__pxgnt_end_int}
      {\PackageError{pxGNT}{Chart end date precedes its start}{Correct start and end.}}
  }

\cs_new_protected:Npn \__pxgnt_item:nnn #1#2#3
  {
    \bool_if:NF \l__pxgnt_fixed_label_width_bool
      {
        \hbox_set:Nn \l__pxgnt_label_box {{\footnotesize #1}}
        \dim_set:Nn \l__pxgnt_label_dim
          {
            \dim_max:nn {\l__pxgnt_label_dim}
              {\dim_min:nn {\l__pxgnt_label_max_dim}
                {\box_wd:N \l__pxgnt_label_box + \labelsep}}
          }
      }
    \seq_gput_right:Nn \g__pxgnt_items_seq
      {\__pxgnt_bar:nnn{#1}{#2}{#3}}
  }

\cs_new_protected:Npn \__pxgnt_bar:nnn #1#2#3
  {
    \__pxgnt_parse_date:nN {#2} \l__pxgnt_item_start_int
    \__pxgnt_parse_date:nN {#3} \l__pxgnt_item_end_int
    \bool_if:nT
      {
        \int_compare_p:n {\l__pxgnt_item_end_int >= \l__pxgnt_window_start_int}
        &&
        \int_compare_p:n {\l__pxgnt_item_start_int <= \l__pxgnt_window_end_int}
      }
      {
        \__pxgnt_format_date:nN
          {\int_max:nn {\l__pxgnt_item_start_int}{\l__pxgnt_window_start_int}}
          \l__pxgnt_bar_start_date_tl
        \__pxgnt_format_date:nN
          {\int_min:nn {\l__pxgnt_item_end_int}{\l__pxgnt_window_end_int}}
          \l__pxgnt_bar_end_date_tl
        \bool_if:NF \l__pxgnt_first_bool {\ganttnewline}
        \bool_set_false:N \l__pxgnt_first_bool
        \int_incr:N \l__pxgnt_row_int
        \int_if_odd:nTF {\l__pxgnt_row_int}
          {
            \ganttbar[
              bar/.append~style={fill=primaryColor!65!secondaryColor,draw=primaryColor}
            ]{#1}{\tl_use:N \l__pxgnt_bar_start_date_tl}
                 {\tl_use:N \l__pxgnt_bar_end_date_tl}
          }
          {
            \ganttbar[
              bar/.append~style={fill=secondaryColor!75,draw=primaryColor!70}
            ]{#1}{\tl_use:N \l__pxgnt_bar_start_date_tl}
                 {\tl_use:N \l__pxgnt_bar_end_date_tl}
          }
      }
  }

\cs_new_protected:Npn \__pxgnt_chart:
  {
    \__pxgnt_format_date:nN {\l__pxgnt_window_start_int}
      \l__pxgnt_window_start_date_tl
    \__pxgnt_format_date:nN {\l__pxgnt_window_end_int}
      \l__pxgnt_window_end_date_tl
    \int_set:Nn \l__pxgnt_window_days_int
      {\l__pxgnt_window_end_int-\l__pxgnt_window_start_int+1}
    \bool_if:NTF \l__pxgnt_fixed_day_width_bool
      {\dim_set_eq:NN \l__pxgnt_day_dim \l__pxgnt_requested_day_dim}
      {\dim_set:Nn \l__pxgnt_day_dim
        {(\l__pxgnt_chart_width_dim - \l__pxgnt_label_dim - 2\labelsep)
          / \l__pxgnt_window_days_int}}
    \par\noindent
    {\small\bfseries\color{primaryColor}
       \tl_use:N \l__pxgnt_window_start_date_tl
       ~--~\tl_use:N \l__pxgnt_window_end_date_tl\par}
    \smallskip
    \noindent
    \begin{ganttchart}[
      time~slot~format=little-endian,
      time~slot~unit=day,
      x~unit=\l__pxgnt_day_dim,
      y~unit~chart=.9\baselineskip,
      y~unit~title=.75\baselineskip,
      hgrid={*1{draw=primaryColor!15}},
      vgrid={*1{draw=primaryColor!18}},
      canvas/.append~style={fill=secondaryColor!6},
      title/.append~style={fill=primaryColor!9,draw=primaryColor!40},
      title~label~font=\scriptsize\bfseries\color{primaryColor},
      bar~label~font=\footnotesize\color{primaryColor},
      bar~label~node/.append~style={text~width=\l__pxgnt_label_dim,align=right,inner~sep=0pt},
      bar~height=.57
    ]{\tl_use:N \l__pxgnt_window_start_date_tl}
     {\tl_use:N \l__pxgnt_window_end_date_tl}
      \gantttitlecalendar{year,month=shortname,day}\ganttnewline
      \bool_set_true:N \l__pxgnt_first_bool
      \int_zero:N \l__pxgnt_row_int
      \seq_map_inline:Nn \g__pxgnt_items_seq {##1}
    \end{ganttchart}
    \par\bigskip
  }

\cs_new_protected:Npn \__pxgnt_render:
  {
    \dim_set:Nn \l__pxgnt_chart_width_dim
      {\bool_if:NTF \l__pxgnt_rotate_bool {\textheight} {\linewidth}}
    \bool_if:NT \l__pxgnt_fixed_day_width_bool
      {
        \int_set:Nn \l__pxgnt_fit_days_int
          {\fp_to_int:n
            {floor(
              \dim_to_fp:n {\l__pxgnt_chart_width_dim - \l__pxgnt_label_dim - 2\labelsep}
              / \dim_to_fp:n {\l__pxgnt_requested_day_dim})}}
        \int_compare:nNnT {\l__pxgnt_fit_days_int} < {1}
          {\PackageError{pxGNT}{day-width cannot fit alongside labels}
             {Reduce day-width or label-width.}}
        \int_set:Nn \l__pxgnt_days_int
          {\int_min:nn {\l__pxgnt_days_int}{\int_max:nn {1}{\l__pxgnt_fit_days_int}}}
      }
    \int_set:Nn \l__pxgnt_window_count_int
      {\int_div_truncate:nn
        {\l__pxgnt_end_int-\l__pxgnt_start_int+\l__pxgnt_days_int}
        {\l__pxgnt_days_int}}
    \int_set:Nn \l__pxgnt_days_int
      {\int_div_truncate:nn
        {\l__pxgnt_end_int-\l__pxgnt_start_int+\l__pxgnt_window_count_int}
        {\l__pxgnt_window_count_int}}
    \int_set_eq:NN \l__pxgnt_window_start_int \l__pxgnt_start_int
    \int_while_do:nn
      {\l__pxgnt_window_start_int <= \l__pxgnt_end_int}
      {
        \int_set:Nn \l__pxgnt_window_end_int
          {\int_min:nn
            {\l__pxgnt_window_start_int+\l__pxgnt_days_int-1}
            {\l__pxgnt_end_int}}
        \bool_if:NTF \l__pxgnt_rotate_bool
          {\begin{sidewaysfigure}[p]\centering\__pxgnt_chart:\end{sidewaysfigure}}
          {\__pxgnt_chart:}
        \int_set:Nn \l__pxgnt_window_start_int
          {\l__pxgnt_window_end_int+1}
      }
  }
