\documentclass{article}

% Load pxCORE with debug enabled, libraries disabled (unit-test style).
\usepackage[debug,PRP=false,UML=false,SRC=false]{pxCORE}

\begin{document}
\ExplSyntaxOn
% Core configuration keys exist only where the target package has a key setup.
\clist_map_inline:nn {PRPconfig,LSTconfig,MEMconfig}
  {\keys_if_exist:nnT {pxCORE}{#1}
    {\PackageError{core-test}{Unsupported configuration key accepted: #1}{}}}
\ExplSyntaxOff

Debug should be ON:
\pxCOREIfDebugTF{[DEBUG-ON]}{[DEBUG-OFF]}

\pxCOREsetup{debug=false}

Debug should now be OFF:
\pxCOREIfDebugTF{[DEBUG-ON-2]}{[DEBUG-OFF-2]}

\begingroup
\pxCOREsetup{debug=true}
\pxCOREIfDebugTF{}{\PackageError{core-test}{Local debug setting failed}{}}
\endgroup
\pxCOREIfDebugTF{\PackageError{core-test}{Debug setting leaked out of its group}{}}{}

\end{document}
