%% Minimal version of a preamble loader, it sets the subject information and expects the assessment short and full name.
%% #1 Assessment short name (mandatory)
%% #2 Assessment full name (mandatory)
%% #3 P3CTeX options (optional, default empty)
%% Usage: \LoadPreamble{PR1}{Primera Practica de la Evaluacio Continua}[PAC, no-plagi]
\NewDocumentCommand{\LoadPreamble}{m m O{}}{
	\PECTeXconfig{
		data= {
			subj-fullname	= {Minimal Template},
			subj-shortname	= {MT},
			subj-code		= {05.597},
			ca-fullname		= {#2},
			ca-shortname	= {#1},
			date			= {\today}
		},
		PR, %SET PR AS DEFAULT DESIGN, STILL CAN BE OVERWRITTEN OR SIMPLY REMOVE THIS LINE
		#3
	}
}
